```37:42:learned_v3_additive.py
def score_additive(pe_pool, q, M, lam):
    """W = cosine + λ·ReLU(peᵀ M q). pe_pool (K,d), q (d,)."""
    peu = _unit(pe_pool)
    qu = q / max(np.linalg.norm(q), 1e-12)
    cos = peu @ qu
    resid = peu @ M @ qu
    return cos + lam * np.maximum(0.0, resid)
```
